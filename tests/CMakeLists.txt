# Standard single-source tests. Tests with fixtures or extra arguments remain explicit.
function(ember_test name)
  add_executable(test_${name} test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE ember)
  add_test(NAME ${name} COMMAND test_${name})
endfunction()
